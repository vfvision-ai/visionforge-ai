## Model Explainability

### Grad-CAM Visualization

```python
from core.explainability import GradCAM, explain_prediction
import torch
import cv2
import numpy as np

# Option 1: High-level interface
model = ...  # Your trained model
image = cv2.imread("path/to/image.jpg")

results = explain_prediction(
    model=model,
    image=image,
    method='gradcam',
    target_class=None  # Auto-detect predicted class
)

# Option 2: Low-level interface
gradcam = GradCAM(model, target_layer='layer4')  # Specify layer
input_tensor = torch.randn(1, 3, 224, 224)
heatmap = gradcam.generate_cam_pytorch(input_tensor)

# Visualize
overlay = gradcam.visualize(image, heatmap, alpha=0.4)
cv2.imwrite("gradcam_result.jpg", overlay)
```

## Model Versioning & Experiment Tracking

### Register a Model

```python
from core.versioning import ModelRegistry, ModelVersion, ModelStage
from pathlib import Path

# Initialize registry
registry = ModelRegistry("./model_registry")

# Create model version metadata
model_version = ModelVersion(
    name="resnet50_cifar10",
    version="1.0.0",
    framework="pytorch",
    architecture="ResNet50",
    stage=ModelStage.DEVELOPMENT,
    created_at=datetime.now().isoformat(),
    updated_at=datetime.now().isoformat(),
    metrics={
        "accuracy": 0.92,
        "f1_score": 0.91,
        "loss": 0.15
    },
    hyperparameters={
        "learning_rate": 0.001,
        "batch_size": 32,
        "epochs": 50
    },
    dataset_hash="abc123def456",
    model_hash="model_hash_here",
    description="Initial ResNet50 model for CIFAR-10"
)

# Register model
model_path = Path("./models/resnet50_v1.0.0.pth")
model_id = registry.register_model(model_version, model_path)
print(f"Model registered: {model_id}")
```

### Track Experiments

```python
from core.versioning import ExperimentTracker, ExperimentStatus

# Initialize tracker
tracker = ExperimentTracker("./experiments")

# Create experiment
exp_id = tracker.create_experiment(
    name="resnet50_cifar10_exp1",
    dataset_path="/data/cifar10",
    model_architecture="ResNet50",
    framework="pytorch",
    hyperparameters={
        "learning_rate": 0.001,
        "batch_size": 32,
        "optimizer": "adam"
    },
    tags=["cifar10", "resnet", "baseline"]
)

# Start experiment
tracker.start_experiment(exp_id)

# During training, log metrics
tracker.log_metrics(exp_id, {"loss": 0.5, "accuracy": 0.85}, step=10)

# Complete experiment
tracker.complete_experiment(
    exp_id,
    metrics={
        "final_accuracy": 0.92,
        "best_loss": 0.12
    },
    artifacts={
        "model": "./models/best_model.pth",
        "plots": "./plots/training_curve.png"
    }
)

# Find best experiment
best_exp = tracker.get_best_experiment("final_accuracy", higher_is_better=True)
print(f"Best experiment: {best_exp.name} with accuracy {best_exp.metrics['final_accuracy']}")
```

### Promote Model to Production

```python
# Promote to production
registry.promote_model("resnet50_cifar10", "1.0.0", ModelStage.PRODUCTION)

# Compare versions
comparison = registry.compare_versions("resnet50_cifar10", "1.0.0", "1.1.0")
print(f"Metrics diff: {comparison['metrics_diff']}")
```

## Security Features

### Rate Limiting

```python
from utils.security import get_rate_limiter, RateLimitConfig

# Configure rate limiter
config = RateLimitConfig(
    max_requests=100,  # 100 requests
    window_seconds=60,  # per minute
    burst_size=120     # allow bursts up to 120
)

limiter = get_rate_limiter("api_endpoints", config)

# Check if request is allowed
user_id = "user_123"
if limiter.is_allowed(user_id):
    # Process request
    print("Request allowed")
else:
    print("Rate limit exceeded")
    
# Get remaining requests
remaining = limiter.get_remaining(user_id)
print(f"Remaining requests: {remaining}")
```

### API Key Management

```python
from utils.security import get_api_key_manager
from pathlib import Path

# Initialize API key manager
key_manager = get_api_key_manager(Path("./api_keys"))

# Generate API key
api_key = key_manager.generate_key(
    user_id="user_123",
    name="Production API Key",
    expires_days=365,
    permissions=["read", "write", "train"]
)
print(f"Generated API key: {api_key}")

# Validate API key
if key_manager.validate_key(api_key, required_permission="train"):
    print("API key valid")
else:
    print("Invalid or insufficient permissions")

# Revoke API key
key_manager.revoke_key(api_key)
```

### Request Signature Validation

```python
from utils.security import RequestSignatureValidator
import time

# Initialize validator
validator = RequestSignatureValidator(secret_key="your-secret-key")

# Sign request
timestamp = int(time.time())
signature = validator.sign_request(
    method="POST",
    path="/api/train",
    body='{"model": "resnet50"}',
    timestamp=timestamp
)

# Validate signature
is_valid = validator.validate_signature(
    signature=signature,
    method="POST",
    path="/api/train",
    body='{"model": "resnet50"}',
    timestamp=timestamp,
    max_age_seconds=300
)
print(f"Signature valid: {is_valid}")
```

## Model Compression

### Quantization

```python
from core.compression import ModelCompressor
import torch

# Initialize compressor
model = ...  # Your trained model
compressor = ModelCompressor(model, framework='pytorch')

# Dynamic quantization (easiest, no calibration needed)
quantized_model = compressor.quantize(method='dynamic', dtype='int8')

# Static quantization (requires calibration data)
calibration_data = [torch.randn(1, 3, 224, 224) for _ in range(100)]
quantized_model = compressor.quantize(
    method='static',
    dtype='int8',
    calibration_data=calibration_data
)

# Check model size
original_size = compressor.get_model_size()
print(f"Original size: {original_size['size_mb']:.2f} MB")

compressor_q = ModelCompressor(quantized_model, 'pytorch')
quantized_size = compressor_q.get_model_size()
print(f"Quantized size: {quantized_size['size_mb']:.2f} MB")
print(f"Compression ratio: {original_size['size_mb'] / quantized_size['size_mb']:.2f}x")
```

### Model Pruning

```python
# Prune 50% of parameters
pruned_model = compressor.prune(
    pruning_ratio=0.5,
    method='magnitude',
    structured=False
)

# Structured pruning (entire filters)
pruned_model = compressor.prune(
    pruning_ratio=0.3,
    method='magnitude',
    structured=True
)
```

## Monitoring & Alerting

### System Monitoring

```python
from utils.monitoring import get_system_monitor, get_metrics_collector

# Get metrics collector
metrics = get_metrics_collector()

# Start system monitoring
monitor = get_system_monitor()
monitor.start_monitoring(interval_seconds=10)

# Record custom metrics
metrics.record('training_loss', 0.15, labels={'model': 'resnet50'})
metrics.record('gpu_utilization', 85.5, unit='percent')

# Get latest metric
latest_loss = metrics.get_latest('training_loss')
print(f"Latest loss: {latest_loss.value}")

# Get metric history
history = metrics.get_history('training_loss', seconds=3600)
print(f"Loss samples in last hour: {len(history)}")

# Get average
avg_loss = metrics.get_average('training_loss', seconds=600)
print(f"Average loss (10 min): {avg_loss}")

# Export for Prometheus
prometheus_metrics = metrics.export_prometheus()
print(prometheus_metrics)
```

### Alert Rules

```python
from utils.monitoring import get_alert_manager, AlertLevel

# Get alert manager
alert_manager = get_alert_manager()

# Add alert rules
alert_manager.add_rule(
    name="high_gpu_temp",
    metric_name="gpu_temperature_celsius",
    threshold=80,
    operator='>',
    level=AlertLevel.CRITICAL,
    message_template="GPU temperature is {value}°C (threshold: {threshold}°C)"
)

alert_manager.add_rule(
    name="low_disk_space",
    metric_name="system_disk_percent",
    threshold=90,
    operator='>',
    level=AlertLevel.WARNING,
    message_template="Disk usage at {value}%"
)

# Start checking
alert_manager.start_checking(interval_seconds=30)

# Get active alerts
active_alerts = alert_manager.get_active_alerts()
for alert in active_alerts:
    print(f"[{alert.level.value}] {alert.message}")
```

```python
from utils.monitoring import get_health_checker
```

## Data Versioning

### Version Control for Datasets

```python
from core.data_versioning import DataVersionControl, DatasetStage
from pathlib import Path

# Initialize DVC
dvc = DataVersionControl(Path("./data_versions"))

# Commit a dataset version
dataset_version = dvc.commit(
    name="cifar10",
    dataset_path=Path("./data/cifar10"),
    version="1.0.0",
    stage=DatasetStage.RAW,
    metadata={
        "num_classes": 10,
        "image_size": [32, 32],
        "source": "official"
    },
    tags=["baseline", "raw"],
    description="Original CIFAR-10 dataset"
)

# Create derived version with transformations
processed_version = dvc.commit(
    name="cifar10",
    dataset_path=Path("./data/cifar10_augmented"),
    version="1.1.0",
    stage=DatasetStage.PROCESSED,
    parent_version="1.0.0",
    transformations=["random_flip", "color_jitter", "normalize"],
    metadata={
        "num_classes": 10,
        "augmentation": "enabled"
    },
    description="CIFAR-10 with augmentation"
)

# List all versions
versions = dvc.list_versions("cifar10")
for v in versions:
    print(f"{v.name} v{v.version} - {v.stage.value}")

# Compare versions
diff = dvc.diff("cifar10", "1.0.0", "1.1.0")
print(f"Size difference: {diff['size_diff_bytes']} bytes")
print(f"Sample count difference: {diff['samples_diff']}")

# Promote to production
dvc.promote_version("cifar10", "1.1.0", DatasetStage.PRODUCTION)

# Tag version
dvc.tag_version("cifar10", "1.1.0", "best-performer")
```

### Data Lineage Tracking

```python
from core.data_versioning import LineageTracker
from pathlib import Path

# Initialize lineage tracker
tracker = LineageTracker(Path("./lineage"))

# Record lineage
lineage = tracker.record_lineage(
    dataset_name="cifar10_combined",
    dataset_version="1.0.0",
    source_datasets=[
        ("cifar10", "1.0.0"),
        ("cifar100", "1.0.0")
    ],
    transformations=[
        {"type": "merge", "method": "concatenate"},
        {"type": "augment", "methods": ["flip", "rotate"]},
        {"type": "normalize", "mean": [0.5, 0.5, 0.5]}
    ],
    created_by="user_123"
)

# Trace lineage upstream
lineage_chain = tracker.trace_upstream("cifar10_combined", "1.0.0")
for l in lineage_chain:
    print(f"Dataset: {l.dataset_name} v{l.dataset_version}")
    print(f"Sources: {l.source_datasets}")

# Generate lineage graph (DOT format)
dot_graph = tracker.visualize_lineage("cifar10_combined", "1.0.0")
print(dot_graph)
```
